## 0. 本课做什么

一句话：**一条 `odp-validate` 命令，对 D3 产出的数据集做 15 项只读质检，按严重度聚合成退出码——ERROR 拦停（2）、WARNING 提醒（1）、PASS/INFO 放行（0）。**

```
configs/datasets/demo.yaml ──┐
runs/data_pipeline/<run_id>/manifest.json ──┤
data/processed/demo/ ──┐     │
                       ▼     ▼
              DatasetSnapshot（只读快照）
                       │
              15 个 check（注册表自动发现）
                       │
              ValidationReport ──→ logger + json + csv
                       │
              退出码:ERROR→2 / WARNING→1 / PASS·INFO→0
```

三条硬规则：

1. **质检只读**：任何 check 不得改数据；发现问题→人去修源头（或重跑 D3），再质检；
2. **判断分级，不二元**：ERROR=必须修，WARNING=可训但建议修，INFO=提示，PASS=通过；退出码取全场最严；
3. **契约对账不互信**：D4 不 import D3 的任何模块，只认公共契约 `common/lineage.py`——拿 manifest 重算指纹、逐样本核验 sha256。

**前置**：D3 已完成，`configs/datasets/demo.yaml` + manifest 存在。本课新增：

## 2. `data_validation/registry.py`：严重度 + 结果 + 注册表

与 D3 同一注册表模式（模块级 dict + 装饰器自注册 + 懒加载自动发现），加格式/策略/质检项都是"加一个文件"。

```python
# apps/platform/src/od_platform/data_validation/registry.py
"""质检注册表 + 严重度 + 结果类型 + 执行上下文。

严重度四级:ERROR(必须修)> WARNING(可训但建议修)> INFO(提示)> PASS(通过);
rank() 供聚合取"全场最严"。重复注册直接 ValueError —— 静默覆盖等于埋雷。
"""
from __future__ import annotations

from dataclasses import dataclass, field
from pathlib import Path
from typing import Any, Callable, Dict, List

from od_platform.data_validation.snapshot import DatasetSnapshot


class CheckSeverity:
    ERROR = "ERROR"
    WARNING = "WARNING"
    INFO = "INFO"
    PASS = "PASS"

    _ORDER = {ERROR: 3, WARNING: 2, INFO: 1, PASS: 0}

    @classmethod
    def rank(cls, severity: str) -> int:
        return cls._ORDER[severity]


@dataclass
class CheckResult:
    name: str
    severity: str
    summary: str                                     # 一句话结论(给人看)
    details: Dict[str, Any] = field(default_factory=dict)  # 结构化证据(给程序看)

    @property
    def passed(self) -> bool:
        return self.severity in (CheckSeverity.PASS, CheckSeverity.INFO)


@dataclass
class CheckContext:
    """一次质检运行共享的上下文。"""
    yaml_path: Path
    snapshot: DatasetSnapshot


CheckFunc = Callable[[CheckContext], CheckResult]


@dataclass(frozen=True)
class CheckEntry:
    name: str
    func: CheckFunc


_REGISTRY: Dict[str, CheckEntry] = {}
_LAZY_INITIALIZED = False


def check(name: str):
    """装饰器:把函数注册为质检项。"""
    def deco(func: CheckFunc) -> CheckFunc:
        if name in _REGISTRY:
            raise ValueError(f"质检项重复注册: {name!r}")
        _REGISTRY[name] = CheckEntry(name=name, func=func)
        return func
    return deco


def _lazy_init() -> None:
    global _LAZY_INITIALIZED
    if _LAZY_INITIALIZED:
        return
    from od_platform.common.registry_utils import import_submodules
    from od_platform.data_validation import checks
    import_submodules(checks)
    _LAZY_INITIALIZED = True


def get_all_checks() -> List[CheckEntry]:
    _lazy_init()
    return [_REGISTRY[k] for k in sorted(_REGISTRY)]


def get_check(name: str) -> CheckEntry:
    _lazy_init()
    return _REGISTRY[name]          # 名字写错就该炸(KeyError)


def list_check_names() -> List[str]:
    _lazy_init()
    return sorted(_REGISTRY)
```

## 3. `snapshot.py`：数据集快照（加载数据，不下结论）

快照是全部 check 的共享输入，原则：**只如实呈现，不做判断**（判断是 check 的事）。两个层次：

- **结构层**：yaml 解析结果、数据集根、类名表、各 split 的图像清单/应有标签路径/实有标签文件、扫描告警；
- **内容层**：每个 split 的 `SplitStats` 有界聚合（图数/有标注图数/实例数/各类实例数）——统计类 check 零 I/O 消费；逐框明细不设上限、不进快照，几何类 check 需要时按需重读文件。

两套标签清单别搞混：`labels_per_split` = 按图像清单**期望存在**的标签路径（用于查缺失）；`label_files_per_split` = 磁盘上**实际存在**的 .txt（用于解析内容、查孤儿）。

```python
# apps/platform/src/od_platform/data_validation/snapshot.py
"""数据集快照:把 yaml + 磁盘状态装进一个只读数据结构,供所有 check 共享。
装数据的，不下结论 —— snapshot 只负责"如实呈现",判断是 check 的事。
"""
from __future__ import annotations

import logging
from dataclasses import dataclass, field
from pathlib import Path
from typing import Dict, List, Optional, Tuple

import yaml

from od_platform.common.constants import IMAGE_EXTENSIONS, Task
from od_platform.data_validation.labels import parse_label_line

logger = logging.getLogger(__name__)

_SPLITS = ("train", "val", "test")


@dataclass(frozen=True)
class SplitStats:
    """单个 split 的有界聚合统计(内存占用与样本数无关)。"""
    image_count: int = 0
    annotated_count: int = 0                        # 有 ≥1 个有效框的图数
    instance_count: int = 0                         # 有效框总数
    class_instances: Dict[int, int] = field(default_factory=dict)


@dataclass
class DatasetSnapshot:
    """一次质检运行共享的数据集快照。"""
    yaml_path: Path
    yaml_data: Dict = field(default_factory=dict)
    yaml_load_error: Optional[str] = None
    data_root: Optional[Path] = None
    nc: Optional[int] = None
    class_names: List[str] = field(default_factory=list)
    task_type: str = Task.DETECT
    images_per_split: Dict[str, Tuple[Path, ...]] = field(default_factory=dict)
    labels_per_split: Dict[str, Tuple[Path, ...]] = field(default_factory=dict)        # 期望存在
    label_files_per_split: Dict[str, Tuple[Path, ...]] = field(default_factory=dict)   # 磁盘实有
    stats_per_split: Dict[str, SplitStats] = field(default_factory=dict)
    scan_warnings: List[str] = field(default_factory=list)

    @property
    def splits(self) -> List[str]:
        return sorted(self.images_per_split)

    @property
    def total_images(self) -> int:
        return sum(len(v) for v in self.images_per_split.values())


# ---------------------------------------------------------------- 内部装配

def _load_yaml(yaml_path: Path) -> Tuple[Dict, Optional[str]]:
    try:
        data = yaml.safe_load(yaml_path.read_text(encoding="utf-8"))
    except Exception as e:
        return {}, f"{type(e).__name__}: {e}"
    if not isinstance(data, dict):
        return {}, f"yaml 顶层不是 mapping: {type(data).__name__}"
    return data, None


def _resolve_data_root(yaml_path: Path, data: Dict) -> Optional[Path]:
    raw = data.get("path")
    if not raw:
        return None
    p = Path(raw)
    return p if p.is_absolute() else (yaml_path.parent / p).resolve()


def _list_images(split_dir: Path) -> Tuple[Path, ...]:
    exts = {e.lower() for e in IMAGE_EXTENSIONS} | {e.upper() for e in IMAGE_EXTENSIONS}
    return tuple(sorted(p for p in split_dir.iterdir() if p.suffix in exts))


def _label_path_for_image(img: Path) -> Path:
    """.../images/train/x.jpg → .../labels/train/x.txt(替换最后一段 images)。"""
    parts = list(img.parts)
    idx = len(parts) - 1 - parts[::-1].index("images")
    parts[idx] = "labels"
    return Path(*parts).with_suffix(".txt")


def _list_label_files(data_root: Path, split: str) -> Tuple[Path, ...]:
    d = data_root / "labels" / split
    return tuple(sorted(d.glob("*.txt"))) if d.exists() else ()


def _normalize_names(names) -> List[str]:
    if isinstance(names, dict):
        return [names[i] for i in sorted(names)]
    if isinstance(names, list):
        return list(names)
    return []


def _build_stats(snap: DatasetSnapshot) -> None:
    """内容层:扫一遍期望标签路径,聚合 SplitStats。读取失败跳过(结构问题归 label_format)。"""
    nc = snap.nc
    for split, label_paths in snap.labels_per_split.items():
        n_annotated = n_inst = 0
        class_inst: Dict[int, int] = {}
        for lbl in label_paths:
            if not lbl.exists():
                continue
            try:
                text = lbl.read_text(encoding="utf-8")
            except OSError:
                continue
            n_valid = 0
            for line in text.splitlines():
                line = line.strip()
                if not line:
                    continue
                box, kind = parse_label_line(line, snap.task_type)
                if box is None:
                    continue
                n_valid += 1
                cid = box[0]
                if isinstance(nc, int) and 0 <= cid < nc:
                    class_inst[cid] = class_inst.get(cid, 0) + 1
            if n_valid:
                n_annotated += 1
                n_inst += n_valid
        snap.stats_per_split[split] = SplitStats(
            image_count=len(snap.images_per_split.get(split, ())),
            annotated_count=n_annotated,
            instance_count=n_inst,
            class_instances=class_inst)


def build_snapshot(yaml_path: Path, task_type: Optional[str] = None) -> DatasetSnapshot:
    yaml_path = Path(yaml_path)
    snap = DatasetSnapshot(yaml_path=yaml_path)
    data, err = _load_yaml(yaml_path)
    snap.yaml_data, snap.yaml_load_error = data, err
    if err:
        return snap                                     # yaml 都读不了,后续 check 各自早返回

    snap.data_root = _resolve_data_root(yaml_path, data)
    nc = data.get("nc")
    snap.nc = nc if isinstance(nc, int) else None
    snap.class_names = _normalize_names(data.get("names"))

    odp_meta = data.get("odp_meta") or {}
    resolved = task_type or odp_meta.get("task") or Task.DETECT
    if task_type and odp_meta.get("task") and task_type != odp_meta["task"]:
        logger.warning("task_type=%s 与 yaml odp_meta.task=%s 不一致,以参数为准",
                       task_type, odp_meta["task"])
    snap.task_type = resolved

    if snap.data_root is None or not snap.data_root.exists():
        if snap.data_root is not None:
            snap.scan_warnings.append(f"split 根目录不存在: {snap.data_root}")
        return snap

    for split in _SPLITS:
        if split not in data:
            continue
        img_dir = snap.data_root / data[split]
        if not img_dir.exists():
            snap.scan_warnings.append(f"split '{split}' 的图像目录不存在: {img_dir}")
            snap.images_per_split[split] = ()
            snap.labels_per_split[split] = ()
            snap.label_files_per_split[split] = ()
            continue
        images = _list_images(img_dir)
        snap.images_per_split[split] = images
        snap.labels_per_split[split] = tuple(_label_path_for_image(p) for p in images)
        snap.label_files_per_split[split] = _list_label_files(snap.data_root, split)

    _build_stats(snap)
    return snap
```

## 4. `labels.py`：标签行解析的唯一权威

全框架只有这里解析标签行。**解析与判断分离**：它只说"这行是什么"（`kind`），严不严重由 check 决定。

```python
# apps/platform/src/od_platform/data_validation/labels.py
"""YOLO 标签行解析:全框架唯一权威。
返回 (box, kind):box=(class_id, [coords]);kind=None 表示合法;box=None 时 kind 指明原因。
"""
from __future__ import annotations

from typing import List, Optional, Tuple

from od_platform.common.constants import Task

KIND_FIELD_COUNT = "field_count_mismatch"
KIND_PARSE = "parse_error"
KIND_POLY_FEW = "polygon_too_few_points"

ParsedBox = Tuple[int, List[float]]


def parse_label_line(line: str, task_type: str) -> Tuple[Optional[ParsedBox], Optional[str]]:
    """解析一行 YOLO 标签(调用方先 strip;空行返回 (None, None))。"""
    parts = line.split()
    if not parts:
        return None, None
    n = len(parts)
    if task_type == Task.SEGMENT:
        if n < 7:
            return None, KIND_POLY_FEW if n >= 3 else KIND_FIELD_COUNT
        if n % 2 == 0:
            return None, KIND_POLY_FEW
    else:
        if n != 5:
            return None, KIND_FIELD_COUNT
    try:
        class_id = int(parts[0])
        coords = [float(v) for v in parts[1:]]
    except ValueError:
        return None, KIND_PARSE
    return (class_id, coords), None
```

### 5.1 `checks/yaml_schema.py`

```python
# apps/platform/src/od_platform/data_validation/checks/yaml_schema.py
"""check: yaml_schema —— 配置文件本身合法吗(最前置的一道)。
yaml 无法加载是前置错,早返回;结构问题收集齐了一次报全。"""
from __future__ import annotations

from typing import List

from od_platform.data_validation.registry import (
    CheckContext, CheckResult, CheckSeverity, check,
)

_NAME = "yaml_schema"


@check(_NAME)
def validate_yaml_schema(ctx: CheckContext) -> CheckResult:
    if ctx.snapshot.yaml_load_error:
        return CheckResult(_NAME, CheckSeverity.ERROR, "yaml 无法加载",
                           {"error": ctx.snapshot.yaml_load_error})
    data = ctx.snapshot.yaml_data
    problems: List[str] = []
    nc, names = data.get("nc"), data.get("names")
    if not isinstance(nc, int) or nc <= 0:
        problems.append(f"nc 缺失或不是正整数: {nc!r}")
    if not isinstance(names, (list, dict)):
        problems.append(f"names 缺失或类型不对(应为 list/dict): {type(names).__name__}")
    if isinstance(nc, int) and isinstance(names, (list, dict)) and len(names) != nc:
        problems.append(f"nc={nc} 与 names 数量 {len(names)} 不一致")
    if not data.get("path"):
        problems.append("缺少 path 字段(数据集根目录)")
    for s in ("train", "val", "test"):
        if s in data and not isinstance(data[s], str):
            problems.append(f"{s} 字段应为字符串路径: {data[s]!r}")
    if problems:
        return CheckResult(_NAME, CheckSeverity.ERROR,
                           f"yaml 结构有 {len(problems)} 处问题", {"problems": problems})
    return CheckResult(_NAME, CheckSeverity.PASS, "yaml 结构合法", {"nc": nc})
```

### 5.2 `checks/pair_existence.py`

```python
# apps/platform/src/od_platform/data_validation/checks/pair_existence.py
"""check: pair_existence —— 每张图都有对应标签吗(按缺失比例分级:≥50% ERROR / ≥5% WARNING)。"""
from __future__ import annotations

from typing import List

from od_platform.common.constants import (
    PAIR_MAX_DETAIL, PAIR_MISSING_ERROR_RATIO, PAIR_MISSING_WARN_RATIO,
)
from od_platform.data_validation.registry import (
    CheckContext, CheckResult, CheckSeverity, check,
)

_NAME = "pair_existence"


@check(_NAME)
def validate_pair_existence(ctx: CheckContext) -> CheckResult:
    snap = ctx.snapshot
    missing: List[dict] = []
    total = 0
    for split in snap.splits:
        for lbl in snap.labels_per_split[split]:
            total += 1
            if not lbl.exists():
                missing.append({"split": split, "expect": str(lbl)})
    if total == 0:
        return CheckResult(_NAME, CheckSeverity.WARNING, "数据集没有任何图像", {})
    ratio = len(missing) / total
    details = {"total_images": total, "missing": len(missing),
               "missing_ratio": round(ratio, 4),
               "missing_preview": missing[:PAIR_MAX_DETAIL],
               "thresholds": {"error": PAIR_MISSING_ERROR_RATIO,
                              "warn": PAIR_MISSING_WARN_RATIO}}
    if ratio >= PAIR_MISSING_ERROR_RATIO:
        return CheckResult(_NAME, CheckSeverity.ERROR,
                           f"{len(missing)}/{total} 张图缺标签({ratio:.1%})— 疑似导出中断", details)
    if ratio >= PAIR_MISSING_WARN_RATIO:
        return CheckResult(_NAME, CheckSeverity.WARNING,
                           f"{len(missing)}/{total} 张图缺标签({ratio:.1%})", details)
    if missing:
        return CheckResult(_NAME, CheckSeverity.INFO,
                           f"{len(missing)}/{total} 张图缺标签({ratio:.1%})— 多为无目标图,确认即可",
                           details)
    return CheckResult(_NAME, CheckSeverity.PASS, f"全部 {total} 张图均有标签", details)
```

### 5.3 `checks/_lineage.py`（内部件）

```python
# apps/platform/src/od_platform/data_validation/checks/_lineage.py
"""内部工具:从 yaml 的 odp_meta 找到并加载 D3 划分契约。下划线开头,不被自动发现。"""
from __future__ import annotations

from typing import Optional, Tuple

from od_platform.common import paths
from od_platform.common.lineage import SplitManifest


def load_manifest_from_yaml(
    yaml_data: dict,
) -> Tuple[Optional[SplitManifest], Optional[str], dict]:
    """返回 (manifest, error, odp_meta);error 非空即失败原因。"""
    odp_meta = yaml_data.get("odp_meta") or {}
    ref = odp_meta.get("manifest_path")
    if not ref:
        return None, "yaml 缺少 odp_meta.manifest_path(不是 odp-transform 产出?)", odp_meta
    mpath = paths.ROOT_DIR / ref
    if not mpath.exists():
        return None, f"manifest 文件不存在: {mpath}", odp_meta
    try:
        return SplitManifest.read(mpath), None, odp_meta
    except Exception as e:
        return None, f"manifest 读取失败: {type(e).__name__}: {e}", odp_meta
```

```python
# apps/platform/src/od_platform/data_validation/checks/fingerprint_match.py
"""check: fingerprint_match —— 契约指纹三方对账。
① manifest 自洽(重算 == 记录):内容没被改;② yaml 指纹 == manifest 指纹:没张冠李戴。"""
from __future__ import annotations

from od_platform.common.lineage import compute_contract_fingerprint
from od_platform.data_validation.checks._lineage import load_manifest_from_yaml
from od_platform.data_validation.registry import (
    CheckContext, CheckResult, CheckSeverity, check,
)

_NAME = "fingerprint_match"


@check(_NAME)
def validate_fingerprint_match(ctx: CheckContext) -> CheckResult:
    manifest, err, odp_meta = load_manifest_from_yaml(ctx.snapshot.yaml_data)
    if err:
        return CheckResult(_NAME, CheckSeverity.ERROR, f"拿不到 D3 划分契约: {err}", {})
    recomputed = compute_contract_fingerprint(
        manifest.dataset, manifest.strategy, manifest.seed, manifest.ratios,
        manifest.names, manifest.samples, manifest.tool_version)
    if recomputed != manifest.contract_fingerprint:
        return CheckResult(_NAME, CheckSeverity.ERROR,
                           "manifest 自身指纹对不上 —— 内容被改过",
                           {"expected": manifest.contract_fingerprint,
                            "recomputed": recomputed})
    yaml_fp = odp_meta.get("contract_fingerprint")
    if yaml_fp != manifest.contract_fingerprint:
        return CheckResult(_NAME, CheckSeverity.ERROR,
                           "yaml 指纹与 manifest 不一致 —— yaml 张冠李戴",
                           {"yaml_fingerprint": yaml_fp,
                            "manifest_fingerprint": manifest.contract_fingerprint})
    return CheckResult(_NAME, CheckSeverity.PASS, "契约指纹三方一致",
                       {"fingerprint": manifest.contract_fingerprint,
                        "n_samples": len(manifest.samples),
                        "strategy": manifest.strategy, "seed": manifest.seed})
```

### 5.5 `checks/manifest_lineage.py`

```python
# apps/platform/src/od_platform/data_validation/checks/manifest_lineage.py
"""check: manifest_lineage —— 逐样本核验:在契约说的 split 里、内容 sha256 没变。
changed(内容变了)/ moved(split 挪了)/ missing(丢了)分开计数。"""
from __future__ import annotations

import hashlib
from pathlib import Path
from typing import Dict, List, Tuple

from od_platform.common.constants import LINEAGE_MAX_DETAIL
from od_platform.data_validation.checks._lineage import load_manifest_from_yaml
from od_platform.data_validation.registry import (
    CheckContext, CheckResult, CheckSeverity, check,
)

_NAME = "manifest_lineage"


def _sha256(path: Path) -> str:
    return hashlib.sha256(path.read_bytes()).hexdigest()


@check(_NAME)
def validate_manifest_lineage(ctx: CheckContext) -> CheckResult:
    manifest, err, _ = load_manifest_from_yaml(ctx.snapshot.yaml_data)
    if err:
        return CheckResult(_NAME, CheckSeverity.ERROR, f"拿不到 D3 划分契约: {err}", {})
    snap = ctx.snapshot

    # 磁盘实况:(split, stem) → 标签路径;stem → 实际出现的 split 集合
    disk: Dict[Tuple[str, str], Path] = {}
    disk_splits_of_stem: Dict[str, set] = {}
    for split in snap.splits:
        for lbl in snap.label_files_per_split[split]:
            disk[(split, lbl.stem)] = lbl
            disk_splits_of_stem.setdefault(lbl.stem, set()).add(split)

    changed: List[str] = []
    moved: List[dict] = []
    missing: List[str] = []
    for s in manifest.samples:
        lbl = disk.get((s.split, s.stem))
        if lbl is None:
            elsewhere = disk_splits_of_stem.get(s.stem)
            if elsewhere:
                moved.append({"stem": s.stem, "expect": s.split, "actual": sorted(elsewhere)})
            else:
                missing.append(s.stem)
        elif _sha256(lbl) != s.sha256:
            changed.append(s.stem)

    details = {"n_samples": len(manifest.samples),
               "n_changed": len(changed), "n_moved": len(moved), "n_missing": len(missing),
               "changed_preview": changed[:LINEAGE_MAX_DETAIL],
               "moved_preview": moved[:LINEAGE_MAX_DETAIL],
               "missing_preview": missing[:LINEAGE_MAX_DETAIL]}
    if changed or moved or missing:
        return CheckResult(
            _NAME, CheckSeverity.ERROR,
            f"契约核验失败: changed={len(changed)} moved={len(moved)} missing={len(missing)}"
            f" —— 与 D3 产出不一致,重跑 odp-transform 或查谁动了 processed",
            details)
    return CheckResult(_NAME, CheckSeverity.PASS,
                       f"{len(manifest.samples)} 个样本逐条核验通过", details)
```

## 6. `service.py` + `report.py`：执行与报告

```python
# apps/platform/src/od_platform/data_validation/service.py
"""质检服务层:跑全部 check 并聚合。
唯一的 except Exception 护栏在这里:单 check 崩溃降级为 ERROR 结果,不拖垮整场质检。"""
from __future__ import annotations

import time
from datetime import datetime
from pathlib import Path
from typing import List, Optional

from od_platform.common import paths
from od_platform.data_validation.registry import (
    CheckContext, CheckEntry, CheckResult, CheckSeverity, get_all_checks,
)
from od_platform.data_validation.report import ValidationReport
from od_platform.data_validation.snapshot import build_snapshot


def _safe_run_one(entry: CheckEntry, ctx: CheckContext) -> CheckResult:
    try:
        return entry.func(ctx)
    except Exception as e:                          # noqa: BLE001 —— 护栏,必须宽
        return CheckResult(entry.name, CheckSeverity.ERROR,
                           f"check 自身崩溃: {type(e).__name__}: {e}",
                           {"exception": repr(e)})


def run_all_checks(ctx: CheckContext) -> List[CheckResult]:
    return [_safe_run_one(e, ctx) for e in get_all_checks()]


def validate_dataset(yaml_path, task_type: Optional[str] = None,
                     run_id: Optional[str] = None,
                     write_report: bool = True) -> ValidationReport:
    """端到端:快照 → 全量 check → 报告(logger + json/csv)。"""
    t0 = time.perf_counter()
    run_id = run_id or datetime.now().strftime("%Y%m%d-%H%M%S")
    yaml_path = Path(yaml_path)
    snapshot = build_snapshot(yaml_path, task_type=task_type)
    results = run_all_checks(CheckContext(yaml_path=yaml_path, snapshot=snapshot))
    report = ValidationReport(run_id=run_id, snapshot=snapshot, results=results,
                              duration_seconds=time.perf_counter() - t0)
    report.render_to_logger()
    if write_report:
        out_dir = paths.validation_run_dir(run_id)
        out_dir.mkdir(parents=True, exist_ok=True)
        report.write_json(out_dir / "report.json")
        report.write_results_csv(out_dir / "results.csv")
    return report
```

```python
# apps/platform/src/od_platform/data_validation/report.py
"""质检报告:纯数据 + 三种出口(logger / json / csv)。报告不含判断逻辑。"""
from __future__ import annotations

import csv
import json
import logging
from dataclasses import dataclass, field
from pathlib import Path
from typing import Any, Dict, List

from od_platform.data_validation.registry import CheckResult, CheckSeverity

logger = logging.getLogger(__name__)


@dataclass
class ValidationReport:
    run_id: str
    snapshot: Any                                    # DatasetSnapshot(避免循环 import)
    results: List[CheckResult] = field(default_factory=list)
    duration_seconds: float = 0.0

    @property
    def overall_severity(self) -> str:
        """全场最严:按 rank 取最大;空结果视为 PASS。"""
        if not self.results:
            return CheckSeverity.PASS
        return max((r.severity for r in self.results), key=CheckSeverity.rank)

    def to_dict(self) -> Dict[str, Any]:
        return {
            "run_id": self.run_id,
            "yaml_path": str(self.snapshot.yaml_path),
            "overall_severity": self.overall_severity,
            "duration_seconds": round(self.duration_seconds, 3),
            "results": [
                {"name": r.name, "severity": r.severity,
                 "summary": r.summary, "details": r.details}
                for r in self.results
            ],
        }

    def write_json(self, path: Path) -> None:
        path.parent.mkdir(parents=True, exist_ok=True)
        path.write_text(json.dumps(self.to_dict(), ensure_ascii=False, indent=2),
                        encoding="utf-8")

    def write_results_csv(self, path: Path) -> None:
        """utf-8-sig 带 BOM:Excel 直接打开中文不乱码。"""
        path.parent.mkdir(parents=True, exist_ok=True)
        with path.open("w", encoding="utf-8-sig", newline="") as f:
            w = csv.writer(f)
            w.writerow(["run_id", "check", "severity", "summary"])
            for r in self.results:
                w.writerow([self.run_id, r.name, r.severity, r.summary])

    def render_to_logger(self, log=None) -> None:
        log = log or logger
        log.info("=" * 60)
        log.info("数据质检报告 run_id=%s  总评: %s(%.2fs)",
                 self.run_id, self.overall_severity, self.duration_seconds)
        log.info("=" * 60)
        for r in self.results:
            log.info("[%-7s] %s: %s", r.severity, r.name, r.summary)
        fp = next((r for r in self.results if r.name == "fingerprint_match"), None)
        if fp and fp.details.get("fingerprint"):
            log.info("-" * 60)
            log.info("契约指纹: %s", fp.details["fingerprint"])
            log.info("样本数: %s  策略: %s  种子: %s",
                     fp.details.get("n_samples"), fp.details.get("strategy"),
                     fp.details.get("seed"))
```

## 8. 标注内容类 check（4 个）

### 8.1 `checks/label_format.py`

与几何类 check 的分工：这里管"行本身合法不合法"（列数/数值/类别范围/坐标区间），几何 check 管"框相对图像出不出界、有没有意义"。**结构错误的行在这里报一次，几何类 check 跳过不重复报。任何一行坏 → ERROR**（坏行进训练就是炸，没有"比例容忍"）。

```python
# apps/platform/src/od_platform/data_validation/checks/label_format.py
"""check: label_format —— 每行标签的结构与取值合法吗。任何一行坏 → ERROR。"""
from __future__ import annotations

from typing import Dict, List, Optional, Tuple

from od_platform.common.constants import DETAILS_PREVIEW_LIMIT
from od_platform.data_validation.labels import ParsedBox, parse_label_line
from od_platform.data_validation.registry import (
    CheckContext, CheckResult, CheckSeverity, check,
)

_NAME = "label_format"
KIND_CLASS_ID_OUT_OF_RANGE = "class_id_out_of_range"
KIND_COORD_OUT_OF_RANGE = "coord_out_of_range"


def _validate_one_line(line: str, task_type: str,
                       nc: int) -> Tuple[Optional[ParsedBox], Optional[str]]:
    """纯函数:(box, kind);kind=None 即合法。"""
    box, kind = parse_label_line(line, task_type)
    if box is None:
        return None, kind
    class_id, coords = box
    if not (0 <= class_id < nc):
        return None, KIND_CLASS_ID_OUT_OF_RANGE
    if any(v < 0 or v > 1 for v in coords):
        return None, KIND_COORD_OUT_OF_RANGE
    return box, None


@check(_NAME)
def validate_label_format(ctx: CheckContext) -> CheckResult:
    snap = ctx.snapshot
    nc = snap.nc
    if not isinstance(nc, int):
        return CheckResult(_NAME, CheckSeverity.PASS,
                           "nc 不可用,跳过(见 yaml_schema)", {"reason": "nc_unavailable"})
    bad_lines: List[dict] = []
    error_kinds: Dict[str, int] = {}
    n_bad = n_total = 0
    for split in snap.splits:
        for lbl in snap.label_files_per_split[split]:
            try:
                text = lbl.read_text(encoding="utf-8")
            except OSError:
                continue
            for line_no, line in enumerate(text.splitlines(), 1):
                line = line.strip()
                if not line:
                    continue
                n_total += 1
                _box, kind = _validate_one_line(line, snap.task_type, nc)
                if kind is not None:
                    n_bad += 1
                    error_kinds[kind] = error_kinds.get(kind, 0) + 1
                    if len(bad_lines) < DETAILS_PREVIEW_LIMIT:
                        bad_lines.append({"split": split, "label": str(lbl),
                                          "line_no": line_no, "kind": kind,
                                          "line": line})
    if n_bad == 0:
        return CheckResult(_NAME, CheckSeverity.PASS,
                           f"全部 {n_total} 行标签结构与取值合法", {"total_lines": n_total})
    return CheckResult(_NAME, CheckSeverity.ERROR,
                       f"{n_bad}/{n_total} 行标签非法({error_kinds})",
                       {"total_lines": n_total, "n_bad": n_bad,
                        "error_kinds": error_kinds, "bad_lines_preview": bad_lines})
```

### 8.2 `checks/bbox_within_image.py`

出界判断用 `cx±w/2` 合并（左/右/上/下四个方向取最大出界量），容差 `BBOX_EDGE_TOLERANCE`。segment 任务显式 PASS 并说明原因（多边形顶点允许在外）。

```python
# apps/platform/src/od_platform/data_validation/checks/bbox_within_image.py
"""check: bbox_within_image —— 框相对图像出不出界(cx±w/2 合并判断)。
≥10% 出界 → ERROR;有但 <10% → WARNING。结构错误的行跳过(label_format 已报)。"""
from __future__ import annotations

from typing import List

from od_platform.common.constants import (
    BBOX_EDGE_TOLERANCE, BBOX_OUT_ERROR_RATIO, DETAILS_PREVIEW_LIMIT, Task,
)
from od_platform.data_validation.labels import parse_label_line
from od_platform.data_validation.registry import (
    CheckContext, CheckResult, CheckSeverity, check,
)

_NAME = "bbox_within_image"


def _overflow_amount(coords: List[float]) -> float:
    """cx,cy,w,h → 出界量(不出界为 0;容差 BBOX_EDGE_TOLERANCE)。"""
    cx, cy, w, h = coords
    overflow = max(0 - (cx - w / 2), (cx + w / 2) - 1,
                   0 - (cy - h / 2), (cy + h / 2) - 1)
    return max(0.0, overflow - BBOX_EDGE_TOLERANCE)


@check(_NAME)
def validate_bbox_within_image(ctx: CheckContext) -> CheckResult:
    snap = ctx.snapshot
    if snap.task_type == Task.SEGMENT:
        return CheckResult(_NAME, CheckSeverity.PASS,
                           "task=segment,出界检查仅适用 detect(多边形顶点允许在外)",
                           {"reason": "task_not_applicable"})
    out_boxes: List[dict] = []
    n_out = n_total = n_skipped = 0
    for split in snap.splits:
        for lbl in snap.label_files_per_split[split]:
            try:
                text = lbl.read_text(encoding="utf-8")
            except OSError:
                continue
            for line_no, line in enumerate(text.splitlines(), 1):
                line = line.strip()
                if not line:
                    continue
                box, kind = parse_label_line(line, snap.task_type)
                if box is None:
                    n_skipped += 1                     # 结构错误:label_format 已报,跳过
                    continue
                n_total += 1
                if _overflow_amount(box[1]) > 0:
                    n_out += 1
                    if len(out_boxes) < DETAILS_PREVIEW_LIMIT:
                        out_boxes.append({"split": split, "label": str(lbl),
                                          "line_no": line_no, "coords": box[1]})
    details = {"total_boxes": n_total, "n_out": n_out,
               "out_ratio": round(n_out / n_total, 4) if n_total else 0.0,
               "skipped_structure_errors": n_skipped,
               "thresholds": {"error_ratio": BBOX_OUT_ERROR_RATIO,
                              "edge_tolerance": BBOX_EDGE_TOLERANCE},
               "out_boxes_preview": out_boxes}
    if n_total == 0:
        return CheckResult(_NAME, CheckSeverity.PASS, "没有可检查的框", details)
    ratio = n_out / n_total
    if ratio >= BBOX_OUT_ERROR_RATIO:
        return CheckResult(_NAME, CheckSeverity.ERROR,
                           f"{n_out}/{n_total} 个框出界({ratio:.1%})", details)
    if n_out:
        return CheckResult(_NAME, CheckSeverity.WARNING,
                           f"{n_out}/{n_total} 个框出界({ratio:.1%})— 少量,建议修源数据",
                           details)
    return CheckResult(_NAME, CheckSeverity.PASS, f"全部 {n_total} 个框均在图像内", details)
```

### 8.3 `checks/box_geometry.py`

退化框 + 极端宽高比（坐标合法但几何上没有训练意义）。**if-chain 优先级**：一行只报一个问题（先退化、后极端），避免一个坏框被计两次。整体 WARNING——可训，但建议清理。

```python
# apps/platform/src/od_platform/data_validation/checks/box_geometry.py
"""check: box_geometry —— 退化框(零宽/零高/面积极小)+ 极端宽高比。
if-chain 优先级:一行只报一个问题。整体 WARNING:可训但建议清理。仅适用 detect。"""
from __future__ import annotations

from typing import Dict, List, Optional, Tuple

from od_platform.common.constants import (
    DEGENERATE_BOX_MIN_SIZE, DETAILS_PREVIEW_LIMIT, STATS_MAX_ASPECT_RATIO, Task,
)
from od_platform.data_validation.labels import parse_label_line
from od_platform.data_validation.registry import (
    CheckContext, CheckResult, CheckSeverity, check,
)

_NAME = "box_geometry"
KIND_ZERO_WIDTH = "zero_width"
KIND_ZERO_HEIGHT = "zero_height"
KIND_TINY_AREA = "tiny_area"
KIND_EXTREME_AR = "extreme_aspect_ratio"


def _geometry_issue(coords: List[float]) -> Optional[Tuple[str, str]]:
    """(kind, detail) 或 None。if-chain:退化优先于极端宽高比。"""
    _cx, _cy, w, h = coords
    if w <= 0:
        return KIND_ZERO_WIDTH, f"w={w}"
    if h <= 0:
        return KIND_ZERO_HEIGHT, f"h={h}"
    if w < DEGENERATE_BOX_MIN_SIZE or h < DEGENERATE_BOX_MIN_SIZE:
        return KIND_TINY_AREA, f"w={w}, h={h}"
    ratio = max(w / h, h / w)
    if ratio >= STATS_MAX_ASPECT_RATIO:
        return KIND_EXTREME_AR, f"aspect={ratio:.1f}"
    return None


@check(_NAME)
def validate_box_geometry(ctx: CheckContext) -> CheckResult:
    snap = ctx.snapshot
    if snap.task_type != Task.DETECT:
        return CheckResult(_NAME, CheckSeverity.PASS,
                           f"task={snap.task_type},几何检查仅适用 detect",
                           {"reason": "task_not_applicable"})
    issue_kinds: Dict[str, int] = {}
    bad_boxes: List[dict] = []
    total_boxes = n_skipped = 0
    for split in snap.splits:
        for lbl in snap.label_files_per_split[split]:
            try:
                text = lbl.read_text(encoding="utf-8")
            except OSError:
                continue
            for line_no, line in enumerate(text.splitlines(), 1):
                line = line.strip()
                if not line:
                    continue
                box, kind = parse_label_line(line, snap.task_type)
                if box is None:
                    n_skipped += 1
                    continue
                total_boxes += 1
                issue = _geometry_issue(box[1])
                if issue is not None:
                    kind_s, detail = issue
                    issue_kinds[kind_s] = issue_kinds.get(kind_s, 0) + 1
                    if len(bad_boxes) < DETAILS_PREVIEW_LIMIT:
                        bad_boxes.append({"split": split, "label": str(lbl),
                                          "line_no": line_no, "kind": kind_s,
                                          "detail": detail})
    details = {"total_boxes": total_boxes, "issue_kinds": issue_kinds,
               "skipped_structure_errors": n_skipped,
               "thresholds": {"degenerate_min_size": DEGENERATE_BOX_MIN_SIZE,
                              "max_aspect_ratio": STATS_MAX_ASPECT_RATIO},
               "bad_boxes_preview": bad_boxes}
    if not issue_kinds:
        return CheckResult(_NAME, CheckSeverity.PASS,
                           f"全部 {total_boxes} 个框几何正常(无退化框/极端宽高比)", details)
    n_bad = sum(issue_kinds.values())
    return CheckResult(_NAME, CheckSeverity.WARNING,
                       f"{n_bad}/{total_boxes} 个框几何异常({issue_kinds})— 可训但建议清理",
```

```python
# apps/platform/src/od_platform/data_validation/checks/duplicate_annotations.py
"""check: duplicate_annotations —— 完全重复的标签行(规范化后逐行比对)。
完全重复行虚增该类权重 → WARNING。近似重复(高 IoU)不在本课范围(details 注明)。"""
from __future__ import annotations

from collections import Counter
from typing import List

from od_platform.common.constants import DETAILS_PREVIEW_LIMIT
from od_platform.data_validation.registry import (
    CheckContext, CheckResult, CheckSeverity, check,
)

_NAME = "duplicate_annotations"


@check(_NAME)
def validate_duplicate_annotations(ctx: CheckContext) -> CheckResult:
    snap = ctx.snapshot
    dup_rows = 0
    files_with_dup = 0
    previews: List[dict] = []
    for split in snap.splits:
        for lbl in snap.label_files_per_split[split]:
            try:
                lines = [l.strip() for l in lbl.read_text(encoding="utf-8").splitlines()
                         if l.strip()]
            except OSError:
                continue
            counts = Counter(lines)
            dups = {l: n for l, n in counts.items() if n > 1}
            if dups:
                files_with_dup += 1
                dup_rows += sum(n - 1 for n in dups.values())
                if len(previews) < DETAILS_PREVIEW_LIMIT:
                    previews.append({"split": split, "label": str(lbl), "dup": dups})
    details = {"dup_rows": dup_rows, "files_with_dup": files_with_dup,
               "previews": previews,
               "note": "近似重复(高 IoU)未在本课范围,如需请另做 embedding/IoU 方案"}
    if dup_rows == 0:
        return CheckResult(_NAME, CheckSeverity.PASS, "无完全重复的标注行", details)
    return CheckResult(_NAME, CheckSeverity.WARNING,
                       f"{dup_rows} 行完全重复(分布在 {files_with_dup} 个文件)— 会虚增权重",
                       details)
```

### 9.4 `checks/split_presence.py`

兑付 snapshot 的 `scan_warnings`（扫描时没资格下结论，这里下结论）：train 不可用 → ERROR；未声明 val → WARNING；未声明 test → INFO；split 规模过小 → INFO。

```python
# apps/platform/src/od_platform/data_validation/checks/split_presence.py
"""check: split_presence —— yaml 声明的划分真的可用吗(兑付 snapshot 的 scan_warnings)。
train 缺失 ERROR;未声明 val WARNING;未声明 test INFO;split 图数 <30 INFO。整体取最严。"""
from __future__ import annotations

from typing import Any, Dict, List

from od_platform.common.constants import SPLIT_MIN_IMAGES
from od_platform.data_validation.registry import (
    CheckContext, CheckResult, CheckSeverity, check,
)

_NAME = "split_presence"


@check(_NAME)
def validate_split_presence(ctx: CheckContext) -> CheckResult:
    snap = ctx.snapshot
    findings: List[Dict[str, Any]] = []
    for w in snap.scan_warnings:
        if "split" in w:
            findings.append({"severity": CheckSeverity.WARNING, "message": w})
    declared = [s for s in ("train", "val", "test") if s in snap.yaml_data]
    if "train" not in snap.splits:
        findings.append({"severity": CheckSeverity.ERROR,
                         "message": "train 划分不可用 — 无训练数据,流程必须终止"})
    if "val" not in declared:
        findings.append({"severity": CheckSeverity.WARNING,
                         "message": "yaml 未声明 val 划分 — 训练将盲飞(无验证集)"})
    if "test" not in declared:
        findings.append({"severity": CheckSeverity.INFO,
                         "message": "yaml 未声明 test 划分 — 无最终诚实评估"})
    for s in snap.splits:
        n = len(snap.images_per_split[s])
        if n < SPLIT_MIN_IMAGES:
            findings.append({"severity": CheckSeverity.INFO,
                             "message": f"{s} 划分共 {n} 张,低于 {SPLIT_MIN_IMAGES},"
                                        f"评估方差偏大(提示)"})
    if not findings:
        return CheckResult(_NAME, CheckSeverity.PASS,
                           "train/val/test 三划分齐备且规模足够", {})
    severity = max((f["severity"] for f in findings), key=CheckSeverity.rank)
    summary = next(f["message"] for f in findings if f["severity"] == severity)
    return CheckResult(_NAME, severity, summary, {"findings": findings})
```

### 9.6 `checks/class_presence.py`

每个声明的类别，在该出现的划分里出现了吗：全集 0 实例 → INFO（死类别）；val/test 有而 train 没有 → WARNING（模型学不到，评估必然失真）；train 有而 val 没有 → WARNING（训练效果不可评估）。

```python
# apps/platform/src/od_platform/data_validation/checks/class_presence.py
"""check: class_presence —— 每个声明的类别,在该出现的划分里出现了吗。"""
from __future__ import annotations

from typing import Any, Dict, List

from od_platform.data_validation.registry import (
    CheckContext, CheckResult, CheckSeverity, check,
)

_NAME = "class_presence"


def _count(snap, split: str, cid: int) -> int:
    st = snap.stats_per_split.get(split)
    return st.class_instances.get(cid, 0) if st else 0


@check(_NAME)
def validate_class_presence(ctx: CheckContext) -> CheckResult:
    snap = ctx.snapshot
    nc = snap.nc
    if not isinstance(nc, int):
        return CheckResult(_NAME, CheckSeverity.PASS,
                           "nc 不可用,跳过类别出现性检查(见 yaml_schema)",
                           {"reason": "nc_unavailable"})
    findings: List[Dict[str, Any]] = []
    for cid in range(nc):
        name = snap.class_names[cid] if cid < len(snap.class_names) else str(cid)
        n_tr = _count(snap, "train", cid)
        n_va = _count(snap, "val", cid)
        n_te = _count(snap, "test", cid)
        if n_tr + n_va + n_te == 0:
            findings.append({"kind": "declared_unused", "severity": CheckSeverity.INFO,
                             "class": name,
                             "message": f"类别 {name!r} 全集 0 实例 — 类别表与标注脱节(死类别)"})
        elif "train" in snap.splits and n_tr == 0:
            findings.append({"kind": "missing_in_train", "severity": CheckSeverity.WARNING,
                             "class": name,
                             "message": f"类别 {name!r} 在 val/test 出现 {n_va + n_te} 次"
                                        f"但 train 为 0 — 模型无法学习,评估必然失真"})
        elif "val" in snap.splits and n_tr > 0 and n_va == 0:
            findings.append({"kind": "missing_in_val", "severity": CheckSeverity.WARNING,
                             "class": name,
                             "message": f"类别 {name!r} train 有 {n_tr} 但 val 为 0"
                                        f" — 该类训练效果不可评估"})
    if not findings:
        return CheckResult(_NAME, CheckSeverity.PASS,
                           f"{nc} 个类别在各划分均有出现", {"nc": nc})
    severity = max((f["severity"] for f in findings), key=CheckSeverity.rank)
    summary = next(f["message"] for f in findings if f["severity"] == severity)
    return CheckResult(_NAME, severity, summary, {"findings": findings})
```

### 9.7 `checks/class_balance.py`

长尾量化提醒，**只提醒、不开药方**（怎么处理是训练侧的决定）。

```python
# apps/platform/src/od_platform/data_validation/checks/class_balance.py
"""check: class_balance —— 类别失衡(长尾)量化提醒。只提醒,不开药方。
失衡比 = 最多类实例数/最少类实例数(只看非零类);≥50 WARNING / ≥10 INFO。"""
from __future__ import annotations

from typing import Any, Dict, List

from od_platform.common.constants import (
    CLASS_IMBALANCE_INFO_RATIO, STATS_MAX_IMBALANCE_RATIO,
)
from od_platform.data_validation.registry import (
    CheckContext, CheckResult, CheckSeverity, check,
)

_NAME = "class_balance"


@check(_NAME)
def validate_class_balance(ctx: CheckContext) -> CheckResult:
    snap = ctx.snapshot
    findings: List[Dict[str, Any]] = []
    for split, st in snap.stats_per_split.items():
        nonzero = {c: n for c, n in st.class_instances.items() if n > 0}
        if len(nonzero) < 2:
            continue
        max_c = max(nonzero, key=nonzero.get)
        min_c = min(nonzero, key=nonzero.get)
        ratio = nonzero[max_c] / nonzero[min_c]
        if ratio >= STATS_MAX_IMBALANCE_RATIO:
            sev = CheckSeverity.WARNING
        elif ratio >= CLASS_IMBALANCE_INFO_RATIO:
            sev = CheckSeverity.INFO
        else:
            continue
        findings.append({"split": split, "ratio": round(ratio, 1), "severity": sev,
                         "message": f"{split} 失衡比 {ratio:.1f}x — 长尾提醒,"
                                    f"训练时建议关注 class_weights / 重采样"})
    if not findings:
        return CheckResult(_NAME, CheckSeverity.PASS, "各类别实例数未见显著失衡", {})
    severity = max((f["severity"] for f in findings), key=CheckSeverity.rank)
    summary = next(f["message"] for f in findings if f["severity"] == severity)
    return CheckResult(_NAME, severity, summary, {"findings": findings})
```

## 12. 收尾速查

### 12.1 15 项 check 总表

| check | 查什么 | 分级 |
|---|---|---|
| `yaml_schema` | nc/names/path/split 字段结构 | 有问题 → ERROR |
| `pair_existence` | 图→标签缺失比 | ≥50% ERROR / ≥5% WARNING / 有 → INFO |
| `fingerprint_match` | 契约指纹三方一致 | 不一致 → ERROR |
| `manifest_lineage` | 逐样本 sha256 + split 位置 | 有 changed/moved/missing → ERROR |
| `label_format` | 行结构/类别范围/坐标区间 | 任何一行坏 → ERROR |
| `bbox_within_image` | 框出界比（detect） | ≥10% ERROR / 有 → WARNING |
| `box_geometry` | 退化框/极端宽高比（detect） | 有 → WARNING |
| `duplicate_annotations` | 完全重复行 | 有 → WARNING |
| `annotation_coverage` | 无标注图占比 | ≥30% WARNING / ≥5% INFO |
| `orphan_labels` | 有标签没图 | ≥10% WARNING / 有 → INFO |
| `stem_collision` | 同 split 同名多扩展名 | 有 → WARNING |
| `split_presence` | 划分齐备与规模 | train 缺 ERROR / 余 WARNING·INFO |
| `split_uniqueness` | 跨 split 泄漏 | 有 → ERROR（无分级） |
| `class_presence` | 类别该在的划分都在吗 | 死类别 INFO / 缺 train·val WARNING |
| `class_balance` | 长尾失衡比 | ≥50 WARNING / ≥10 INFO |

### 12.2 关键约定

| 约定 | 内容 |
|---|---|
| 质检只读 | 不改任何文件；修数据回源头，修完重跑 D3 |
| 退出码 | ERROR→2 / WARNING→1 / PASS·INFO→0（取全场最严） |
| 解析唯一权威 | `labels.py` 的 `parse_label_line`；判断归 check |
| 结构错不重复报 | `label_format` 报过的行，几何类 check 跳过（details 里记 skipped 数） |
| 快照分层 | 结构层 + SplitStats 有界聚合；逐框明细不设限、按需重读 |
| 阈值进 constants | check 里不写魔法数 |
| 契约不互信 | 只认 `common/lineage.py`；重算指纹 + 逐样本 sha256 对账 |
| 单 check 崩溃 | service 唯一的宽 `except` 降级为 ERROR 结果，不拖垮整场 |

### 12.3 常用排错

| 症状 | 原因与处理 |
|---|---|
| `拿不到 D3 划分契约` | yaml 不是 `odp-transform` 产出，或 manifest 被删——重跑 D3 |
| `manifest 自身指纹对不上` | manifest.json 被手改——重跑 D3 重出契约 |
| `yaml 指纹与 manifest 不一致` | yaml 是从别的数据集拷来的（张冠李戴）——重跑 D3 |
| csv 用 Excel 打开乱码 | 已用 `utf-8-sig`；若仍乱码，检查是不是自己重写过 csv |

```diff
+@contextmanager
+def progress_scope(callback: ProgressCallback | None = None):
+    """每次运行隔离回调，嵌套调用保留外层回调，异常时也恢复上下文。"""
+    token = _callback.set(callback or _callback.get())
+    try:
+        yield
+    finally:
+        _callback.reset(token)

+
+def emit_progress(stage: str, completed: int = 0, total: int = 0) -> None:
+    callback = _callback.get()

+        callback(
+            {
+                "stage": stage,
+                "completed": completed,
+                "total": total,
+                "percent": round(completed / total * 100, 1) if total else 0.0,
+            }
+        )
+
+
+def track(items: Iterable[T], stage: str, total: int | None = None) -> Iterator[T]:
+    """完成一次工作后再递增；限制回调频率，避免大数据集被 UI 更新拖慢。"""
+    if total is None:
+        total = len(items) if hasattr(items, "__len__") else 0
+    emit_progress(stage, 0, total)
+    last = time.monotonic()
+    completed = 0
+    for item in items:
+        yield item
+        completed += 1
+        now = time.monotonic()
+        if now - last >= 0.1:
+            emit_progress(stage, completed, total)
+            last = now
+    emit_progress(stage, completed, total)
```

```diff
+                    card.scroll_into_view_if_needed()
+                    if args.screenshot:
+                        args.screenshot.parent.mkdir(parents=True, exist_ok=True)
+                        card.screenshot(path=str(args.screenshot))
+                    page.set_viewport_size({"width": 390, "height": 844})
+                    assert page.evaluate(
+                        "document.documentElement.scrollWidth <= window.innerWidth"
+                    )
+                    assert not errors, errors
+                    browser.close()
+                    print(
+                        "Browser checks passed: D3, D4, progress, new/fixed issues, mobile layout."
+                    )
+            finally:
+                server.shutdown()
+                server.server_close()
+                thread.join(timeout=5)
+                from django.db import connections
+
+                connections.close_all()
+
+
+if __name__ == "__main__":
+    main()
```
